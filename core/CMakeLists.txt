cmake_minimum_required(VERSION 3.18)
project(libsisp VERSION 1.0.0 DESCRIPTION "SISP assessment core (portable C11)" LANGUAGES C)
include(GNUInstallDirs)

option(SISP_BUILD_SHARED "Build the shared library" ON)
option(SISP_BUILD_TESTS "Build unit and parity tests" ON)
option(SISP_BUILD_EXAMPLES "Build the sisp-cli example" ON)
option(SISP_WERROR "Treat warnings as errors" OFF)
option(SISP_BUILD_STORE "Build sisp_store, the optional SQLite history store (vendored SQLite)" ON)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_C_VISIBILITY_PRESET hidden)

set(SISP_SOURCES src/sisp.c src/sisp_flat.c)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  set(SISP_WARNINGS -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wstrict-prototypes)
  if(SISP_WERROR)
    list(APPEND SISP_WARNINGS -Werror)
  endif()
elseif(MSVC)
  set(SISP_WARNINGS /W4)
endif()


function(sisp_configure target)
  target_include_directories(${target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_compile_options(${target} PRIVATE ${SISP_WARNINGS})
  target_compile_definitions(${target} PRIVATE SISP_BUILDING)
  if(UNIX)
    target_link_libraries(${target} PUBLIC m)
  endif()
endfunction()

add_library(sisp_static STATIC ${SISP_SOURCES})
set_target_properties(sisp_static PROPERTIES OUTPUT_NAME sisp POSITION_INDEPENDENT_CODE ON)
sisp_configure(sisp_static)
add_library(sisp::sisp_static ALIAS sisp_static)

if(SISP_BUILD_SHARED)
  add_library(sisp_shared SHARED ${SISP_SOURCES})
  set_target_properties(sisp_shared PROPERTIES OUTPUT_NAME sisp VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR})
  target_compile_definitions(sisp_shared PUBLIC SISP_SHARED)
  sisp_configure(sisp_shared)
  add_library(sisp::sisp ALIAS sisp_shared)
endif()

if(SISP_BUILD_EXAMPLES)
  add_executable(sisp-cli examples/sisp_cli.c)
  target_link_libraries(sisp-cli PRIVATE sisp_static)
  target_compile_options(sisp-cli PRIVATE ${SISP_WARNINGS})
endif()

if(SISP_BUILD_TESTS)
  enable_testing()
  add_executable(test_unit tests/test_unit.c)
  target_link_libraries(test_unit PRIVATE sisp_static)
  add_test(NAME unit COMMAND test_unit)

  add_executable(test_parity tests/test_parity.c)
  target_link_libraries(test_parity PRIVATE sisp_static)
  add_test(NAME parity_with_typescript COMMAND test_parity ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/parity.txt)

  if(SISP_BUILD_SHARED)
    add_executable(test_unit_shared tests/test_unit.c)
    target_link_libraries(test_unit_shared PRIVATE sisp_shared)
    add_test(NAME unit_shared COMMAND test_unit_shared)
  endif()
  if(SISP_BUILD_EXAMPLES)
    add_test(NAME cli_assess COMMAND ${CMAKE_COMMAND} -DCLI=$<TARGET_FILE:sisp-cli> -DCSV=${CMAKE_CURRENT_SOURCE_DIR}/examples/checkins.csv -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_assess.cmake)
  endif()
endif()

if(SISP_BUILD_STORE)
  add_subdirectory(store)
endif()

install(TARGETS sisp_static EXPORT sispTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(SISP_BUILD_SHARED)
  install(TARGETS sisp_shared EXPORT sispTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(FILES include/sisp.h include/sisp_flat.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT sispTargets FILE sispConfig.cmake NAMESPACE sisp:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/sisp)
include(CMakePackageConfigHelpers)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/sispConfigVersion.cmake COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/sispConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/sisp)
